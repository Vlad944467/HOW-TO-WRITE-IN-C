Это гайд как сделать командную строку на языке программирования C.

1. Сначало создаем файл например shell.c
2. После того как мы создали файл нам надо добавить базовые
библиотеки С :


    #include <stdio.h>
    #include <string.h>

    //Это базовые библиотеки C.

3. Сделаем точку входа, без точки входа будет ошибка:

   //Пишем:

       int main() {

       }

   //Отлично, мы создали точку входа. Точка входа
   //обязательно должна быть названа main иначе будет ошибка.
   //В этих скобках мы и будем писать код.

4. Добавляем цикл:

//теперь мы пишем:

       char cmd[50];

    // Здесь cmd это название переменной с типом char.
    // В квадратных скобках число 50 это количество символов которых
    // можно максимум написать, вы можете заменить на 60 или 40 и так далее.


    //Пишем в скобки main:

        while (1) {

        }

    //Хорошо, мы создали цикл while.
    //В скобках while будет сама командная строка.

5. Делаем командную строку:

    //Теперь нам надо написать в скобках цикла while:

       fgets(cmd, sizeof(cmd), stdin);


    //Теперь нам надо перейти к if/else
    //Например напишем:

        if (strcmp(cmd, "word") == 0) {
            printf("hello\n");
        }
    //Там где в ковычках я написал word, там вы можете написать свое
    //название команды, после того как вы напишите название команды в командную строку
    //командная строка выведет результат.
    //В скобках мы видим printf это вывод в консоль, а \n это перенос на другую строку.

    //Если захотите сделать еще одну команду то пишите уже else if, а не просто if.
    //Еще желательно написать else:

        else {
            printf("unknown command\n");
        }
    //Если команда будет неправильно написана или такой команды не будет существовать
    //то командная строка выведет unknown command.

6. Завершение:

   //Теперь нам надо написать:

       return 0;

   //Это успешное завершение программы.

   //Обязательно надо написать этот return 0; после закрывание
   //фигурных скобок цикла while.


/*
    Теперь вы написали минимальную командную строку на C.
    Вы можете добавлять свои команды с помощью else if.
*/

7. Полный код:

#include <stdio.h>
#include <string.h>

int main() {
    char cmd[50];

    while (1) {
        printf("> ");
        fgets(cmd, sizeof(cmd), stdin);
        cmd[strcspn(cmd, "\n")] = 0;  // убираем \n

        if (strcmp(cmd, "word") == 0) {
            printf("hello\n");
        }
        else if (strcmp(cmd, "exit") == 0) {
            break;
        }
        else {
            printf("unknown command\n");
        }
    }
    return 0;
}

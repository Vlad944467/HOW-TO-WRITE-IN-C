Как писать цикл for:

Разберем классический пример:

    for (int i = 0; i < 10; i++) {
        printf("hello\n");
    }

    /*
        В этом примере мы видим цикл for.
        int i = 0; — объявление переменной i со значением 0.
        i < 10; — условие: цикл повторяется 10 раз.
        i++ — увеличение i на 1 после каждой итерации.
    */
    
Теперь разберем второй пример:

    int i = 0;
    
    for (; i < 10 ;) {
        printf("hello\n");
        i++;
    }
    
    /*
        Здесь результат тот же, но код выглядит иначе.
        Переменная i объявлена до цикла.
        i++ теперь внутри фигурных скобок.
    */
    
Теперь пример с вводом и выводом:

    char cmd[15];

    printf("input: ");    
    fgets(cmd, sizeof(cmd), stdin);
    
    for (int i = 0; i < 10; i++) {
        printf("%s", cmd);   // %s для строки
    }
    
Тот же пример, но с вынесенной переменной:

    char cmd[15];
    int i = 0;
    
    for (; i < 10 ;) {
        printf("%s", cmd);
        i++;
    }

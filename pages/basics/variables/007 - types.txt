Что такое типы переменных:
    
    //У переменных есть разные типы.
    //Если вы не укажите тип то будет ошибка
    
    //Типы:
    
    //---------------------
       
       //Чтоб добавить переменной тип просто
       //пишем тип и потом название переменной
       

       //Тип int нужен для целых чисел:
       
       int a = 6;
       

       //Тип float нужен для нецелых чисел
       //В типе float максимум ~6-7 цифр после запятой:
       
       float price = 5.9999999f;
       
       double price2 = 5.1234567899;
       
       //Тип char нужен для одного символа:
       
       char s = 'h';
       

       //и также тип char нужен для массива:
       
       char name[20];


    //---------------------
    
    //Таблица типов:
    
    
    //=========================================
    Тип     Размер      Пример

    int     4 байта    int age = 25;
    float   4 байта    float pi = 3.14f;
    double  8 байт     double pi = 3.141592;
    char    1 байт     char letter = 'A';
    char[]  N байт     char name[20] = "Alex";
    //=========================================

Продолжение о перменных:

    /*
     В прошлом файле с номером 008 я
     не рассказал все о переменных,
     и это продолжение.
    */


    //В C есть еще модификаторы типов

    //Пример:

    //-------------------
        //обычный int

        int a;  //Обычный int может хранить отрицательные числа.

        //А тип int с модификатором unsigned может хранить
        //только положительные числа:

        unsigned int a;

        //он помещает в 2 раза больше
        //положительных чисел
        //чем обычный int.

    /-------------------

    /*
     Также существует модификатор const
     для констант, он не изменяется
     то есть если вы измените значение
     то компилятор выдаст ошибку.
    */

    //Пример:

    //-----------------------------------

        //Обычный тип
        //покажу на типе int

        int b = 2; //Значение можно изменять.

        //int с модификатором const

        const int b = 2;

        //значение после объявления не изменяется
        //с модификатором const

    //-----------------------------------

    //Также есть еще static:


    //------------------------------------
    /*
     Без static обычная переменная
     просто она создаётся заново каждый раз
    */

    int x = 0;   //создается каждый раз заново.

    /*
     А с static переменная создаётся
     один раз и сохраняет значение:
    */

    static int x = 0;

    //-------------------------------------

    //Мы разобрали модификаторы типов в C.

Как работать с условными оператороми if / else:

    //Условные операторы if/else легкие, но очень
    //важные в программировании.

    //If / else это основа программирования.
    //Без неё программа это просто
    //Прямая линия без выбора.
    //А if / else дает программе мозги.


    //Разберем пример:
    
    //-----------------------------
    
       int ch;
       scanf("%d", &ch);  // ← исправлено: &ch

       if (ch == 1) {
          return 0;
       }
       else if (ch == 2) {
        // здесь код для команды 2
       }
       else {
           printf("unknown command\n");
       }
       return 0;
       }
    
    //-----------------------------
    
    //Я уже показывал похожие примеры
    //Мы создали переменную ch с типом
    //int.
    
    //Дальше мы видим что оператор if
    //сравнивает выбор с цифрой 1
    //если мы напишем цифру 3 то
    //то программа выведет unknown command
    //так как не одного совпадания нету.
    //В скобках вы можете написать что ходите
    //я вам показал просто шаблон.
    
    //Также мы можем дописывать свои команды
    //используя else if.
    
    
    //Но я вам показал пример с числами
    //и в примере с числами мы
    //используем scanf, а со строками
    //мы будем использовать fgets.
    
    
    //Пример со строками:
    
    //------------------------------
    
       char ch[20];
       fgets(ch, sizeof(ch), stdin);
    
       if (strcmp(ch, "exit") == 0) {
           return 0;
       }
       else if (strcmp(ch, "word") == 0) {
    
       }
       else {
           printf("unknown command\n");
       }
    
    //-------------------------------
    
    //Это тот же самый шаблон, но только
    //он сравнивает строки, а не как в
    //прошлом шаблоне цифры.
    //strcmp - это сравнение строк.
    
    
    //Полный код шаблонов:
    
    //Шаблон с цифрами:
    
    //----------------------------------
    
       #include <stdio.h>
       #include <string.h>

       int main() {
          int ch;
          scanf("%d", &ch);  // ← исправлено: &ch

          if (ch == 1) {
             return 0;
          }
          else if (ch == 2) {
            // здесь код для команды 2
          }
          else {
              printf("unknown command\n");
          }
          return 0;
       }
    
    //-----------------------------------
    

    //Шаблон со строками:
    
    //-----------------------------------
    
       #include <stdio.h>
       #include <string.h>
    
       int main () {
           char ch[20];
           fgets(ch, sizeof(ch), stdin);
    
           if (strcmp(ch, "exit") == 0) {
               return 0;
           }
           else if (strcmp(ch, "word") == 0) {
    
           }
           else {
               printf("unknown command\n");
           }
           return 0;
       }
    
    //-----------------------------------
    
    /*
     ЗАПОМНИ:
     if — "если условие верно"
     else if — "иначе если другое условие верно"
     else — "если ничего не подошло"

     ДЛЯ СТРОК используй strcmp()!
     ДЛЯ ЧИСЕЛ используй ==, !=, <, >, <=, >=
    */
